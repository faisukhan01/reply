     1→"use client";
     2→
     3→import { useEffect, useRef, useState } from "react";
     4→import { useParams } from "next/navigation";
     5→import { motion, AnimatePresence } from "framer-motion";
     6→import { Bot, Send, Minus, MessageSquare, AlertCircle, Star, Heart, Headphones, DollarSign, Clock, Check, CheckCheck } from "lucide-react";
     7→import { cn } from "@/lib/utils";
     8→
     9→type WidgetConfig = {
    10→  id: string;
    11→  name: string;
    12→  welcomeMessage: string;
    13→  primaryColor: string;
    14→  status: string;
    15→  position?: string;
    16→};
    17→
    18→type ChatMsg = {
    19→  id: string;
    20→  role: "VISITOR" | "AI";
    21→  content: string;
    22→  timestamp?: number;
    23→  read?: boolean;
    24→};
    25→
    26→const VISITOR_ID_KEY = "replyai_visitor_id";
    27→const VISITOR_NAME_KEY = "replyai_visitor_name";
    28→
    29→function getOrCreateVisitorId(): string {
    30→  if (typeof window === "undefined") {
    31→    return "v-" + Math.random().toString(36).slice(2, 12);
    32→  }
    33→  try {
    34→    let id = window.localStorage.getItem(VISITOR_ID_KEY);
    35→    if (!id) {
    36→      id =
    37→        "v-" +
    38→        Math.random().toString(36).slice(2, 12) +
    39→        Date.now().toString(36);
    40→      window.localStorage.setItem(VISITOR_ID_KEY, id);
    41→    }
    42→    return id;
    43→  } catch {
    44→    return "v-" + Math.random().toString(36).slice(2, 12);
    45→  }
    46→}
    47→
    48→function getStoredVisitorName(): string {
    49→  if (typeof window === "undefined") return "";
    50→  try {
    51→    return window.localStorage.getItem(VISITOR_NAME_KEY) || "";
    52→  } catch {
    53→    return "";
    54→  }
    55→}
    56→
    57→function storeVisitorName(name: string) {
    58→  if (typeof window === "undefined") return;
    59→  try {
    60→    window.localStorage.setItem(VISITOR_NAME_KEY, name);
    61→  } catch {
    62→    /* ignore */
    63→  }
    64→}
    65→
    66→const QUICK_ACTIONS = [
    67→  { label: "Pricing", message: "I'd like to know about pricing", icon: DollarSign },
    68→  { label: "Hours", message: "What are your business hours?", icon: Clock },
    69→  { label: "Talk to human", message: "I'd like to speak with a human agent", icon: Headphones },
    70→];
    71→
    72→export default function WidgetPage() {
    73→  const params = useParams<{ botId: string }>();
    74→  const botId = params?.botId;
    75→
    76→  const [config, setConfig] = useState<WidgetConfig | null>(null);
    77→  const [paused, setPaused] = useState(false);
    78→  const [notFound, setNotFound] = useState(false);
    79→  const [loading, setLoading] = useState(true);
    80→
    81→  const [messages, setMessages] = useState<ChatMsg[]>([]);
    82→  const [input, setInput] = useState("");
    83→  const [sending, setSending] = useState(false);
    84→  const [widgetOpen, setWidgetOpen] = useState(true);
    85→  const [conversationId, setConversationId] = useState<string | null>(null);
    86→  const [showSurvey, setShowSurvey] = useState(false);
    87→  const [surveySubmitted, setSurveySubmitted] = useState(false);
    88→  const [hoverRating, setHoverRating] = useState(0);
    89→  const [selectedRating, setSelectedRating] = useState(0);
    90→
    91→  // Visitor name flow
    92→  const [visitorName, setVisitorName] = useState("");
    93→  const [nameInput, setNameInput] = useState("");
    94→  const [showNamePrompt, setShowNamePrompt] = useState(false);
    95→
    96→  const scrollRef = useRef<HTMLDivElement>(null);
    97→  const inputRef = useRef<HTMLInputElement>(null);
    98→  const visitorIdRef = useRef<string>("");
    99→
   100→  // Generate / load a stable visitorId once on mount.
   101→  useEffect(() => {
   102→    visitorIdRef.current = getOrCreateVisitorId();
   103→    const stored = getStoredVisitorName();
   104→    if (stored) {
   105→      setVisitorName(stored);
   106→    } else {
   107→      setShowNamePrompt(true);
   108→    }
   109→  }, []);
   110→
   111→  // Fetch the bot's public config.
   112→  useEffect(() => {
   113→    if (!botId) return;
   114→    let cancelled = false;
   115→    (async () => {
   116→      try {
   117→        const res = await fetch(`/api/widget?botId=${botId}`);
   118→        const data = await res.json();
   119→        if (cancelled) return;
   120→        if (data.status === "PAUSED") {
   121→          setPaused(true);
   122→        } else if (data.id) {
   123→          setConfig(data);
   124→          setMessages([
   125→            {
   126→              id: "welcome",
   127→              role: "AI",
   128→              content:
   129→                data.welcomeMessage ||
   130→                "Hi! How can I help you today?",
   131→              timestamp: Date.now(),
   132→              read: true,
   133→            },
   134→          ]);
   135→        } else {
   136→          setNotFound(true);
   137→        }
   138→      } catch (e) {
   139→        console.error("[widget] config fetch failed", e);
   140→        if (!cancelled) setNotFound(true);
   141→      } finally {
   142→        if (!cancelled) setLoading(false);
   143→      }
   144→    })();
   145→    return () => {
   146→      cancelled = true;
   147→    };
   148→  }, [botId]);
   149→
   150→  // Auto-scroll to the latest message.
   151→  useEffect(() => {
   152→    if (scrollRef.current) {
   153→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
   154→    }
   155→  }, [messages, sending, widgetOpen]);
   156→
   157→  // Refocus input after sending.
   158→  useEffect(() => {
   159→    if (widgetOpen && !sending && inputRef.current && !showNamePrompt) {
   160→      inputRef.current.focus();
   161→    }
   162→  }, [sending, widgetOpen, showNamePrompt]);
   163→
   164→  async function handleSend(textOverride?: string) {
   165→    const text = (textOverride ?? input).trim();
   166→    if (!text || sending || !config || paused) return;
   167→
   168→    const now = Date.now();
   169→    const visitorMsg: ChatMsg = {
   170→      id: "v-" + now,
   171→      role: "VISITOR",
   172→      content: text,
   173→      timestamp: now,
   174→      read: false,
   175→    };
   176→    setMessages((prev) => [...prev, visitorMsg]);
   177→    setInput("");
   178→    setSending(true);
   179→
   180→    try {
   181→      const res = await fetch(`/api/widget`, {
   182→        method: "POST",
   183→        headers: { "Content-Type": "application/json" },
   184→        body: JSON.stringify({
   185→          botId,
   186→          visitorId: visitorIdRef.current,
   187→          message: text,
   188→          visitorName: visitorName || undefined,
   189→        }),
   190→      });
   191→      const data = await res.json();
   192→      const replyTime = Date.now();
   193→      const reply: ChatMsg = {
   194→        id: "a-" + replyTime,
   195→        role: "AI",
   196→        content:
   197→          (data && data.reply) ||
   198→          "Sorry, I didn't quite catch that. Could you rephrase?",
   199→        timestamp: replyTime,
   200→        read: true,
   201→      };
   202→      // Mark previous visitor messages as read
   203→      setMessages((prev) =>
   204→        prev.map((m) => (m.role === "VISITOR" && !m.read ? { ...m, read: true } : m))
   205→      );
   206→      setMessages((prev) => [...prev, reply]);
   207→      if (data?.conversationId) {
   208→        setConversationId(data.conversationId);
   209→      }
   210→      // Show survey after 3+ visitor messages (meaningful conversation)
   211→      const visitorMsgCount = messages.filter((m) => m.role === "VISITOR").length;
   212→      if (visitorMsgCount >= 2 && !surveySubmitted) {
   213→        setTimeout(() => setShowSurvey(true), 1500);
   214→      }
   215→    } catch (e) {
   216→      console.error("[widget] send failed", e);
   217→      setMessages((prev) => [
   218→        ...prev,
   219→        {
   220→          id: "err-" + Date.now(),
   221→          role: "AI",
   222→          content:
   223→            "Connection error. Please check your internet and try again.",
   224→        },
   225→      ]);
   226→    } finally {
   227→      setSending(false);
   228→    }
   229→  }
   230→
   231→  function handleKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
   232→    if (e.key === "Enter" && !e.shiftKey) {
   233→      e.preventDefault();
   234→      handleSend();
   235→    }
   236→  }
   237→
   238→  function handleSubmitName() {
   239→    const name = nameInput.trim();
   240→    if (!name) return;
   241→    setVisitorName(name);
   242→    storeVisitorName(name);
   243→    setShowNamePrompt(false);
   244→    setNameInput("");
   245→  }
   246→
   247→  async function submitRating(rating: number) {
   248→    setSelectedRating(rating);
   249→    setSurveySubmitted(true);
   250→    setShowSurvey(false);
   251→    if (conversationId) {
   252→      try {
   253→        await fetch(`/api/conversations/${conversationId}`, {
   254→          method: "PATCH",
   255→          headers: { "Content-Type": "application/json" },
   256→          body: JSON.stringify({ satisfaction: rating }),
   257→        });
   258→      } catch {
   259→        /* ignore */
   260→      }
   261→    }
   262→    setMessages((prev) => [
   263→      ...prev,
   264→      {
   265→        id: "sys-" + Date.now(),
   266→        role: "AI",
   267→        content:
   268→          rating >= 4
   269→            ? "Thank you so much for your feedback! I'm glad I could help. Have a wonderful day! ✨"
   270→            : "Thank you for the feedback. I'll share this with our team so we can improve. Is there anything else I can help with?",
   271→      },
   272→    ]);
   273→  }
   274→
   275→  const primaryColor = config?.primaryColor || "#8b5cf6";
   276→  const botName = config?.name || "Assistant";
   277→  const hasQuickActions = messages.length <= 1 && !sending;
   278→
   279→  // ---------- States: loading / paused / not-found ----------
   280→  if (loading) {
   281→    return (
   282→      <div className="h-screen w-full flex items-center justify-center bg-gradient-to-br from-violet-50 to-fuchsia-50 dark:from-violet-950/30 dark:to-fuchsia-950/30">
   283→        <div className="flex flex-col items-center gap-3">
   284→          <div className="h-12 w-12 rounded-full border-4 border-violet-200 border-t-violet-600 animate-spin" />
   285→          <p className="text-sm text-muted-foreground">Loading assistant…</p>
   286→        </div>
   287→      </div>
   288→    );
   289→  }
   290→
   291→  if (paused) {
   292→    return (
   293→      <div className="h-screen w-full flex items-center justify-center bg-gradient-to-br from-violet-50 to-fuchsia-50 dark:from-violet-950/30 dark:to-fuchsia-950/30 p-6">
   294→        <div className="text-center max-w-xs space-y-3">
   295→          <div className="mx-auto h-14 w-14 rounded-full bg-amber-100 dark:bg-amber-950/40 flex items-center justify-center">
   296→            <AlertCircle className="h-7 w-7 text-amber-600 dark:text-amber-400" />
   297→          </div>
   298→          <h2 className="text-base font-semibold">Assistant is offline</h2>
   299→          <p className="text-sm text-muted-foreground leading-relaxed">
   300→            Our assistant has been temporarily paused. Please try again later or
   301→            email support and we&apos;ll get back to you.
   302→          </p>
   303→        </div>
   304→      </div>
   305→    );
   306→  }
   307→
   308→  if (notFound) {
   309→    return (
   310→      <div className="h-screen w-full flex items-center justify-center bg-gradient-to-br from-violet-50 to-fuchsia-50 dark:from-violet-950/30 dark:to-fuchsia-950/30 p-6">
   311→        <div className="text-center max-w-xs space-y-3">
   312→          <div className="mx-auto h-14 w-14 rounded-full bg-rose-100 dark:bg-rose-950/40 flex items-center justify-center">
   313→            <AlertCircle className="h-7 w-7 text-rose-600 dark:text-rose-400" />
   314→          </div>
   315→          <h2 className="text-base font-semibold">Assistant not found</h2>
   316→          <p className="text-sm text-muted-foreground leading-relaxed">
   317→            We couldn&apos;t find this assistant. Please check the embed code or
   318→            contact the site owner.
   319→          </p>
   320→        </div>
   321→      </div>
   322→    );
   323→  }
   324→
   325→  // ---------- Minimized state ----------
   326→  if (!widgetOpen) {
   327→    const hasMessages = messages.length > 1;
   328→    return (
   329→      <div
   330→        className="h-screen w-full flex items-end justify-end p-4 bg-transparent"
   331→        style={{ ["--bot-color" as string]: primaryColor }}
   332→      >
   333→        <motion.button
   334→          initial={{ scale: 0.8, opacity: 0 }}
   335→          animate={{ scale: 1, opacity: 1 }}
   336→          whileHover={{ scale: 1.08 }}
   337→          whileTap={{ scale: 0.92 }}
   338→          onClick={() => setWidgetOpen(true)}
   339→          aria-label="Open chat"
   340→          className="relative h-14 w-14 rounded-full bg-[var(--bot-color)] text-white shadow-2xl flex items-center justify-center"
   341→        >
   342→          <MessageSquare className="h-6 w-6" />
   343→          {hasMessages && (
   344→            <>
   345→              <span className="absolute -top-0.5 -right-0.5 flex h-4 w-4">
   346→                <span className="absolute inline-flex h-full w-full rounded-full bg-rose-400 opacity-75 animate-ping" />
   347→                <span className="relative inline-flex h-4 w-4 rounded-full bg-rose-500 text-[9px] font-bold items-center justify-center">
   348→                  {messages.filter(m => m.role === "AI" && m.id !== "welcome").length}
   349→                </span>
   350→              </span>
   351→              <span className="absolute inset-0 rounded-full animate-glow-pulse" />
   352→            </>
   353→          )}
   354→        </motion.button>
   355→      </div>
   356→    );
   357→  }
   358→
   359→  // ---------- Active chat state ----------
   360→  return (
   361→    <div
   362→      className="h-screen w-full bg-gradient-to-br from-violet-50 to-fuchsia-50 dark:from-violet-950/30 dark:to-fuchsia-950/30 flex items-stretch sm:items-center sm:justify-center sm:p-4"
   363→      style={{ ["--bot-color" as string]: primaryColor }}
   364→    >
   365→      <motion.div
   366→        initial={{ opacity: 0, y: 20, scale: 0.95 }}
   367→        animate={{ opacity: 1, y: 0, scale: 1 }}
   368→        exit={{ opacity: 0, y: 20, scale: 0.95 }}
   369→        transition={{ type: "spring", stiffness: 300, damping: 25 }}
   370→        className="flex flex-col w-full h-full sm:max-w-md sm:h-[640px] sm:max-h-[90vh] sm:rounded-2xl bg-card sm:shadow-2xl overflow-hidden border sm:border-border"
   371→      >
   372→        {/* Header */}
   373→        <header
   374→          className="flex items-center gap-3 px-4 py-3 text-white shrink-0 bg-[var(--bot-color)]"
   375→          style={{
   376→            backgroundImage: `linear-gradient(135deg, var(--bot-color), color-mix(in srgb, var(--bot-color) 70%, #000))`,
   377→          }}
   378→        >
   379→          <div className="flex h-10 w-10 items-center justify-center rounded-full bg-white/20 backdrop-blur-sm ring-2 ring-white/30 shrink-0">
   380→            <Bot className="h-5 w-5" />
   381→          </div>
   382→          <div className="flex-1 min-w-0">
   383→            <div className="font-semibold text-sm truncate">{botName}</div>
   384→            <div className="flex items-center gap-1.5 text-[11px] text-white/85">
   385→              <span className="relative flex h-2 w-2">
   386→                <span className="absolute inline-flex h-full w-full rounded-full bg-emerald-300 opacity-75 animate-ping" />
   387→                <span className="relative inline-flex h-2 w-2 rounded-full bg-emerald-400" />
   388→              </span>
   389→              Online · replies instantly
   390→            </div>
   391→          </div>
   392→          <button
   393→            onClick={() => setWidgetOpen(false)}
   394→            aria-label="Minimize chat"
   395→            className="h-8 w-8 rounded-full flex items-center justify-center hover:bg-white/20 transition-colors shrink-0"
   396→          >
   397→            <Minus className="h-4 w-4" />
   398→          </button>
   399→        </header>
   400→
   401→        {/* Messages */}
   402→        <div
   403→          ref={scrollRef}
   404→          className="flex-1 overflow-y-auto scroll-thin px-4 py-4 space-y-3 bg-gradient-to-b from-background to-muted/30"
   405→        >
   406→          <AnimatePresence initial={false}>
   407→            {messages.map((m) => (
   408→              <Bubble key={m.id} msg={m} primaryColor={primaryColor} />
   409→            ))}
   410→          </AnimatePresence>
   411→
   412→          {/* Quick actions */}
   413→          <AnimatePresence>
   414→            {hasQuickActions && !showNamePrompt && (
   415→              <motion.div
   416→                initial={{ opacity: 0, y: 8 }}
   417→                animate={{ opacity: 1, y: 0 }}
   418→                exit={{ opacity: 0, y: -8 }}
   419→                transition={{ duration: 0.2 }}
   420→                className="flex flex-wrap gap-2 pt-2"
   421→              >
   422→                {QUICK_ACTIONS.map((action) => {
   423→                  const ActionIcon = action.icon;
   424→                  return (
   425→                    <motion.button
   426→                      key={action.label}
   427→                      onClick={() => handleSend(action.message)}
   428→                      whileHover={{ scale: 1.05 }}
   429→                      whileTap={{ scale: 0.95 }}
   430→                      className="inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium text-white transition-all hover:shadow-lg"
   431→                      style={{ backgroundColor: primaryColor }}
   432→                    >
   433→                      <ActionIcon className="h-3 w-3" />
   434→                      {action.label}
   435→                    </motion.button>
   436→                  );
   437→                })}
   438→              </motion.div>
   439→            )}
   440→          </AnimatePresence>
   441→
   442→          {sending && (
   443→            <motion.div
   444→              initial={{ opacity: 0, y: 8 }}
   445→              animate={{ opacity: 1, y: 0 }}
   446→              className="flex items-end gap-2"
   447→            >
   448→              <div className="h-7 w-7 rounded-full bg-muted flex items-center justify-center shrink-0">
   449→                <Bot className="h-4 w-4 text-muted-foreground" />
   450→              </div>
   451→              <div className="bg-muted rounded-2xl rounded-bl-md px-4 py-3 flex items-center gap-1.5">
   452→                <span className="typing-dot h-2 w-2 rounded-full bg-muted-foreground/70" />
   453→                <span className="typing-dot h-2 w-2 rounded-full bg-muted-foreground/70" />
   454→                <span className="typing-dot h-2 w-2 rounded-full bg-muted-foreground/70" />
   455→              </div>
   456→            </motion.div>
   457→          )}
   458→        </div>
   459→
   460→        {/* Visitor name prompt */}
   461→        <AnimatePresence>
   462→          {showNamePrompt && (
   463→            <motion.div
   464→              initial={{ opacity: 0, height: 0 }}
   465→              animate={{ opacity: 1, height: "auto" }}
   466→              exit={{ opacity: 0, height: 0 }}
   467→              transition={{ duration: 0.2 }}
   468→              className="shrink-0 border-t bg-gradient-to-r from-violet-50 to-fuchsia-50 dark:from-violet-950/30 dark:to-fuchsia-950/20 px-4 py-3 overflow-hidden"
   469→            >
   470→              <div className="text-center">
   471→                <div className="text-sm font-medium text-violet-900 dark:text-violet-100 mb-2">
   472→                  Hi! What&apos;s your name?
   473→                </div>
   474→                <div className="flex items-center gap-2">
   475→                  <input
   476→                    type="text"
   477→                    value={nameInput}
   478→                    onChange={(e) => setNameInput(e.target.value)}
   479→                    onKeyDown={(e) => {
   480→                      if (e.key === "Enter") handleSubmitName();
   481→                    }}
   482→                    placeholder="Your name"
   483→                    className="flex-1 h-9 rounded-full bg-white dark:bg-card px-4 text-sm outline-none focus:ring-2 focus:ring-violet-400 transition-all placeholder:text-muted-foreground"
   484→                    autoFocus
   485→                  />
   486→                  <button
   487→                    onClick={handleSubmitName}
   488→                    disabled={!nameInput.trim()}
   489→                    className="h-9 px-4 rounded-full text-white text-sm font-medium transition-all hover:opacity-90 active:scale-95 disabled:opacity-40"
   490→                    style={{ backgroundColor: primaryColor }}
   491→                  >
   492→                    Go
   493→                  </button>
   494→                </div>
   495→              </div>
   496→            </motion.div>
   497→          )}
   498→        </AnimatePresence>
   499→
   500→        {/* Satisfaction survey */}
   501→        <AnimatePresence>
   502→          {showSurvey && !surveySubmitted && (
   503→            <motion.div
   504→              initial={{ opacity: 0, height: 0 }}
   505→              animate={{ opacity: 1, height: "auto" }}
   506→              exit={{ opacity: 0, height: 0 }}
   507→              transition={{ duration: 0.2 }}
   508→              className="shrink-0 border-t bg-gradient-to-r from-violet-50 to-fuchsia-50 dark:from-violet-950/30 dark:to-fuchsia-950/20 px-4 py-3 overflow-hidden"
   509→            >
   510→              <div className="text-center">
   511→                <div className="text-xs font-medium text-violet-900 dark:text-violet-100 mb-1.5">
   512→                  Rate this conversation
   513→                </div>
   514→                <div className="flex items-center justify-center gap-2">
   515→                  {[1, 2, 3, 4, 5].map((n) => (
   516→                    <button
   517→                      key={n}
   518→                      onClick={() => submitRating(n)}
   519→                      onMouseEnter={() => setHoverRating(n)}
   520→                      onMouseLeave={() => setHoverRating(0)}
   521→                      className="star-interactive p-0.5"
   522→                      aria-label={`Rate ${n} stars`}
   523→                    >
   524→                      <Star
   525→                        className={cn(
   526→                          "h-6 w-6 transition-colors",
   527→                          n <= (hoverRating || selectedRating)
   528→                            ? "fill-amber-400 text-amber-400"
   529→                            : "fill-transparent text-amber-300 hover:text-amber-400"
   530→                        )}
   531→                      />
   532→                    </button>
   533→                  ))}
   534→                </div>
   535→                <button
   536→                  onClick={() => setShowSurvey(false)}
   537→                  className="text-[10px] text-muted-foreground hover:text-foreground mt-1.5"
   538→                >
   539→                  Maybe later
   540→                </button>
   541→              </div>
   542→            </motion.div>
   543→          )}
   544→        </AnimatePresence>
   545→
   546→        {/* Survey thank you (animated heart) */}
   547→        <AnimatePresence>
   548→          {surveySubmitted && (

   616→function formatMsgTime(ts?: number): string {
   617→  if (!ts) return "";
   618→  const now = Date.now();
   619→  const diff = now - ts;
   620→  if (diff < 60000) return "just now";
   621→  if (diff < 3600000) return `${Math.floor(diff / 60000)}m ago`;
   622→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`;
   623→  return new Date(ts).toLocaleTimeString(undefined, { hour: "2-digit", minute: "2-digit" });
   624→}

   627→  msg,
   628→  primaryColor,
   629→  showTimestamp,
   630→}: {
   631→  msg: ChatMsg;
   632→  primaryColor: string;
   633→  showTimestamp?: boolean;

   641→      className={cn(
   642→        "flex items-end gap-2",
   643→        isVisitor ? "justify-end" : "justify-start"
