   110→   110→function formatDate(iso: string) {
   111→   111→  try {
   112→   112→    return new Date(iso).toLocaleDateString(undefined, {
   113→   113→      year: "numeric",
   114→   114→      month: "short",
   115→   115→      day: "numeric",
   116→   116→    });
   117→   117→  } catch {
   118→   118→    return iso;
   119→   119→  }
   120→   120→}

   122→   122→function initials(name: string) {
   123→   123→  return name
   124→   124→    .split(" ")
   125→   125→    .map((p) => p[0])
   126→   126→    .filter(Boolean)
   127→   127→    .slice(0, 2)
   128→   128→    .join("")
   129→   129→    .toUpperCase();
   130→   130→}
