   483→   483→function PreviewWidget({
   484→   484→  botName,
   485→   485→  welcomeMessage,
   486→   486→  primaryColor,
   487→   487→  persona,
   488→   488→}: {
   489→   489→  botName: string;
   490→   490→  welcomeMessage: string;
   491→   491→  primaryColor: string;
   492→   492→  persona: string;
   493→   493→}) {
   494→   494→  const [messages, setMessages] = React.useState<
   495→   495→    { role: "bot" | "user"; text: string }[]
   496→   496→  >([]);
   497→   497→  const [input, setInput] = React.useState("");
   498→   498→  const [sending, setSending] = React.useState(false);
   499→   499→  const scrollRef = React.useRef<HTMLDivElement>(null);
   500→   500→
   501→   501→  React.useEffect(() => {
   502→   502→    scrollRef.current?.scrollTo({
   503→   503→      top: scrollRef.current.scrollHeight,
   504→   504→      behavior: "smooth",
   505→   505→    });
   506→   506→  }, [messages, sending]);
   507→   507→
   508→   508→  async function handleSend() {
   509→   509→    const text = input.trim();
   510→   510→    if (!text || sending) return;
   511→   511→    setInput("");
   512→   512→    setMessages((m) => [...m, { role: "user", text }]);
   513→   513→    setSending(true);
   514→   514→    try {
   515→   515→      const res = await fetch("/api/chatbot/test", {
   516→   516→        method: "POST",
