   234→  const [primaryColor, setPrimaryColor] = React.useState(chatbot.primaryColor);
   235→  const [status, setStatus] = React.useState(chatbot.status);
   236→  const [saving, setSaving] = React.useState(false);
   237→
   238→  // Sync if chatbot is reloaded externally.
   239→  React.useEffect(() => {
   240→    setName(chatbot.name);
   241→    setWelcomeMessage(chatbot.welcomeMessage);
   242→    setPersona(chatbot.persona);
   243→    setSystemPrompt(chatbot.systemPrompt ?? "");
   244→    setPrimaryColor(chatbot.primaryColor);

   276→      onUpdate(data.chatbot as Chatbot);
   277→      toast.success("Chatbot updated", {
   278→        description: "Your changes are live.",
   279→      });
   280→    } catch (e: unknown) {
   281→      const msg = e instanceof Error ? e.message : "Unknown error";
   282→      toast.error("Failed to save", { description: msg });
   283→    } finally {
   284→      setSaving(false);
   285→    }
   286→  }

   306→              id="name"
   307→              value={name}

   310→              maxLength={80}
   311→            />
   312→          </div>

   318→              id="welcomeMessage"
   319→              value={welcomeMessage}

   483→function PreviewWidget({
   484→  botName,
   485→  welcomeMessage,
   486→  primaryColor,
   487→  persona,
   488→}: {
   489→  botName: string;
   490→  welcomeMessage: string;
   491→  primaryColor: string;
   492→  persona: string;
   493→}) {
   494→  const [messages, setMessages] = React.useState<
   495→    { role: "bot" | "user"; text: string }[]
   496→  >([]);
   497→  const [input, setInput] = React.useState("");
   498→  const [sending, setSending] = React.useState(false);
   499→  const scrollRef = React.useRef<HTMLDivElement>(null);
   500→
   501→  React.useEffect(() => {
   502→    scrollRef.current?.scrollTo({
   503→      top: scrollRef.current.scrollHeight,
   504→      behavior: "smooth",
   505→    });
   506→  }, [messages, sending]);
   507→
   508→  async function handleSend() {
   509→    const text = input.trim();
   510→    if (!text || sending) return;
   511→    setInput("");
   512→    setMessages((m) => [...m, { role: "user", text }]);
   513→    setSending(true);
   514→    try {
   515→      const res = await fetch("/api/chatbot/test", {
   516→        method: "POST",
   517→        headers: { "Content-Type": "application/json" },
   518→        body: JSON.stringify({ message: text }),
   519→      });
   520→      if (!res.ok) throw new Error("Request failed");
   521→      const data = await res.json();
   522→      setMessages((m) => [
   523→        ...m,
   524→        { role: "bot", text: data.reply as string },
   525→      ]);
   526→    } catch {
   527→      setMessages((m) => [
   528→        ...m,
   529→        {
   530→          role: "bot",
   531→          text: "I'm having trouble responding right now. Please try again.",
   532→        },
   533→      ]);
   534→    } finally {
   535→      setSending(false);
   536→    }
   537→  }
   538→
   539→  return (
   540→    <Card className="overflow-hidden rounded-xl border shadow-sm">
   541→      <CardHeader
   542→        className="pb-3"
   543→        style={{
   544→          backgroundColor: primaryColor,
   545→          color: "#fff",
   546→        }}
   547→      >
   548→        <div className="flex items-center gap-3">
   549→          <div
   550→            className="flex size-9 items-center justify-center rounded-full bg-white/20 backdrop-blur"
   551→            aria-hidden
   552→          >
   553→            <Bot className="size-5" />
   554→          </div>
   555→          <div className="min-w-0">
   556→            <CardTitle className="truncate text-sm font-semibold">
   557→              {botName}
   558→            </CardTitle>
   559→            <p className="text-xs text-white/80">
   560→              {persona} ·{" "}
   561→              <span className="inline-flex items-center gap-1">
   562→                <span className="inline-block size-1.5 rounded-full bg-emerald-300" />
   563→                Online
   564→              </span>
   565→            </p>
   566→          </div>
   567→        </div>
   568→      </CardHeader>
   569→
   570→      <CardContent className="p-0">
   571→        <div
   572→          ref={scrollRef}
   573→          className="max-h-80 min-h-[220px] space-y-3 overflow-y-auto bg-muted/40 p-4 scroll-thin"
   574→        >
   575→          {/* Welcome bubble */}
   576→          <ChatBubble
   577→            role="bot"
   578→            text={welcomeMessage || "Hi! How can I help you today?"}
   579→            primaryColor={primaryColor}
   580→          />
   581→
   582→          {/* Conversation */}
   583→          {messages.map((m, i) => (
   584→            <ChatBubble
   585→              key={i}
   586→              role={m.role}
   587→              text={m.text}
   588→              primaryColor={primaryColor}
   589→            />
   590→          ))}
   591→
   592→          {/* Typing indicator */}
   593→          {sending && (
   594→            <div className="flex items-start gap-2">
   595→              <div
   596→                className="flex size-7 items-center justify-center rounded-full text-white"
   597→                style={{ backgroundColor: primaryColor }}
   598→                aria-hidden
   599→              >
   600→                <Bot className="size-4" />
   601→              </div>
   602→              <div className="rounded-2xl rounded-bl-sm bg-background px-3 py-2 shadow-sm">
   603→                <div className="flex items-center gap-1">
   604→                  <span className="typing-dot size-1.5 rounded-full bg-muted-foreground" />
   605→                  <span className="typing-dot size-1.5 rounded-full bg-muted-foreground" />
   606→                  <span className="typing-dot size-1.5 rounded-full bg-muted-foreground" />
   607→                </div>
   608→              </div>
   609→            </div>
   610→          )}
   611→        </div>
   612→
   613→        {/* Test input */}
   614→        <div className="border-t bg-background p-3">
   615→          <div className="flex items-center gap-2">
   616→            <Input
   617→              value={input}
   618→              onChange={(e) => setInput(e.target.value)}
   619→              onKeyDown={(e) => {
   620→                if (e.key === "Enter" && !e.shiftKey) {
   621→                  e.preventDefault();
   622→                  handleSend();
   623→                }
   624→              }}
   625→              placeholder="Test your bot…"
   626→              disabled={sending}
   627→              className="h-9"
   628→              aria-label="Test message"
   629→            />
   630→            <Button
   631→              size="icon"
   632→              onClick={handleSend}
   633→              disabled={sending || !input.trim()}
   634→              style={{ backgroundColor: primaryColor, color: "#fff" }}
   635→              aria-label="Send test message"
   636→            >
   637→              {sending ? (
