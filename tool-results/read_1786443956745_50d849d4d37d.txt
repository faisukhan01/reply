   118→function hashString(s: string): number {
   119→  let h = 0;
   120→  for (let i = 0; i < s.length; i++) {
   121→    h = (h << 5) - h + s.charCodeAt(i);
   122→    h |= 0;
   123→  }
   124→  return Math.abs(h);
   125→}
   126→
   127→function avatarColor(name: string): string {
   128→  return AVATAR_COLORS[hashString(name) % AVATAR_COLORS.length];
   129→}
   130→
   131→function initials(name: string | null): string {
   132→  if (!name) return "?";
   133→  const parts = name.trim().split(/\s+/).filter(Boolean);
   134→  if (parts.length === 0) return "?";
   135→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   136→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   137→}

  1395→// ─── MessageList (with grouping) ──────────────────────────────────
  1396→function MessageList({
  1397→  messages,
  1398→  aiTyping,
  1399→}: {
  1400→  messages: Message[];
  1401→  aiTyping: boolean;
  1402→}) {
  1403→  if (messages.length === 0) {
  1404→    return (
  1405→      <div className="flex flex-col items-center justify-center h-full text-center text-muted-foreground py-16">
  1406→        <MessageSquare className="size-10 mb-3 opacity-40" />
  1407→        <p className="text-sm font-medium">No messages yet</p>
  1408→        <p className="text-xs mt-1">
  1409→          Start the conversation by sending a message below.
  1410→        </p>
  1411→      </div>
  1412→    );
  1413→  }
  1414→
  1415→  // Group consecutive messages by same role + < 5 min gap.
  1416→  type Group = { role: Message["role"]; items: Message[] };
  1417→  const groups: Group[] = [];
  1418→  for (const msg of messages) {
  1419→    const last = groups[groups.length - 1];
  1420→    if (
  1421→      last &&
  1422→      last.role === msg.role &&
  1423→      last.items.length > 0 &&
  1424→      new Date(msg.createdAt).getTime() -
  1425→        new Date(last.items[last.items.length - 1].createdAt).getTime() <
  1426→        5 * 60 * 1000
  1427→    ) {
  1428→      last.items.push(msg);
  1429→    } else {
  1430→      groups.push({ role: msg.role, items: [msg] });
  1431→    }
  1432→  }
  1433→
  1434→  return (
  1435→    <div className="space-y-4 max-w-2xl mx-auto">
  1436→      {groups.map((group, gi) => (
  1437→        <MessageGroup key={gi} group={group} />
  1438→      ))}
  1439→      {aiTyping && <TypingBubble />}
  1440→    </div>
  1441→  );
  1442→}
