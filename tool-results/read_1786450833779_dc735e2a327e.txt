   133→function formatDate(iso: string) {
   134→  try {
   135→    return new Date(iso).toLocaleDateString(undefined, {
   136→      year: "numeric",
   137→      month: "short",
   138→      day: "numeric",
   139→    });
   140→  } catch {
   141→    return iso;
   142→  }
   143→}

   145→function initials(name: string) {
   146→  return name
   147→    .split(" ")
   148→    .map((p) => p[0])
   149→    .filter(Boolean)
   150→    .slice(0, 2)
   151→    .join("")
   152→    .toUpperCase();
   153→}

   321→  async function handleCreateWebhook(e: React.FormEvent) {

   325→      toast.error("URL is required");

   329→      new URL(url);

   331→      toast.error("Enter a valid http(s) URL");
   332→      return;
   333→    }
   334→    if (whForm.events.length === 0) {
   335→      toast.error("Select at least one event");
   336→      return;

   339→    try {
   340→      const res = await fetch("/api/webhooks", {
   341→        method: "POST",
   342→        headers: { "Content-Type": "application/json" },
   343→        body: JSON.stringify({
   344→          url,
   345→          events: whForm.events,
   346→          secret: whForm.secret || undefined,
   347→        }),
   348→      });
   349→      if (!res.ok) {
   350→        const j = await res.json().catch(() => null);
   351→        throw new Error(j?.error || "Failed to create webhook");
   352→      }
   353→      const { webhook } = await res.json();
   354→      setWebhooks((prev) => [webhook as Webhook, ...prev]);
   355→      toast.success("Webhook created");

   362→      setCreatingWh(false);

   366→  async function handleToggleWebhook(wh: Webhook, next: boolean) {
   367→    setTogglingId(wh.id);
   368→    setWebhooks((prev) =>
   369→      prev.map((w) => (w.id === wh.id ? { ...w, active: next } : w))
   370→    );
   371→    try {
   372→      const res = await fetch(`/api/webhooks/${wh.id}`, {
   373→        method: "PATCH",
   374→        headers: { "Content-Type": "application/json" },
   375→        body: JSON.stringify({ active: next }),
   376→      });
   377→      if (!res.ok) {
   378→        const j = await res.json().catch(() => null);
   379→        throw new Error(j?.error || "Failed to update webhook");
   380→      }
   381→      const { webhook } = await res.json();
   382→      setWebhooks((prev) =>
   383→        prev.map((w) => (w.id === wh.id ? (webhook as Webhook) : w))
   384→      );
   385→      toast.success(next ? "Webhook enabled" : "Webhook disabled");
   386→    } catch (err: unknown) {
   387→      setWebhooks((prev) =>
   388→        prev.map((w) => (w.id === wh.id ? { ...w, active: !next } : w))
   389→      );
   390→      const msg = err instanceof Error ? err.message : "Failed to update webhook";
   391→      toast.error(msg);
   392→    } finally {
   393→      setTogglingId(null);
   394→    }
   395→  }
