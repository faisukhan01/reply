   124→function hashString(s: string): number {
   125→  let h = 0;
   126→  for (let i = 0; i < s.length; i++) {
   127→    h = (h << 5) - h + s.charCodeAt(i);
   128→    h |= 0;
   129→  }
   130→  return Math.abs(h);
   131→}
   132→
   133→function avatarColor(name: string): string {
   134→  return AVATAR_COLORS[hashString(name) % AVATAR_COLORS.length];
   135→}
   136→
   137→function initials(name: string | null): string {
   138→  if (!name) return "?";
   139→  const parts = name.trim().split(/\s+/).filter(Boolean);
   140→  if (parts.length === 0) return "?";
   141→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   142→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   143→}

  1517→// ─── MessageList (with grouping) ──────────────────────────────────
  1518→function MessageList({
  1519→  messages,
  1520→  aiTyping,
  1521→}: {
  1522→  messages: Message[];
  1523→  aiTyping: boolean;
  1524→}) {
  1525→  if (messages.length === 0) {
  1526→    return (
  1527→      <div className="flex flex-col items-center justify-center h-full text-center text-muted-foreground py-16">
  1528→        <MessageSquare className="size-10 mb-3 opacity-40" />
  1529→        <p className="text-sm font-medium">No messages yet</p>
  1530→        <p className="text-xs mt-1">
  1531→          Start the conversation by sending a message below.
  1532→        </p>
  1533→      </div>
  1534→    );
  1535→  }
  1536→
  1537→  // Group consecutive messages by same role + < 5 min gap.
  1538→  type Group = { role: Message["role"]; items: Message[] };
  1539→  const groups: Group[] = [];
  1540→  for (const msg of messages) {
  1541→    const last = groups[groups.length - 1];
  1542→    if (
  1543→      last &&
  1544→      last.role === msg.role &&
  1545→      last.items.length > 0 &&
  1546→      new Date(msg.createdAt).getTime() -
  1547→        new Date(last.items[last.items.length - 1].createdAt).getTime() <
  1548→        5 * 60 * 1000
  1549→    ) {
  1550→      last.items.push(msg);
  1551→    } else {
  1552→      groups.push({ role: msg.role, items: [msg] });
  1553→    }
  1554→  }
  1555→
  1556→  return (
  1557→    <div className="space-y-4 max-w-2xl mx-auto">
  1558→      {groups.map((group, gi) => (
  1559→        <MessageGroup key={gi} group={group} />
  1560→      ))}
  1561→      {aiTyping && <TypingBubble />}
  1562→    </div>
  1563→  );
  1564→}
