   124→   124→function hashString(s: string): number {
   125→   125→  let h = 0;
   126→   126→  for (let i = 0; i < s.length; i++) {
   127→   127→    h = (h << 5) - h + s.charCodeAt(i);
   128→   128→    h |= 0;
   129→   129→  }
   130→   130→  return Math.abs(h);
   131→   131→}

   133→   133→function avatarColor(name: string): string {
   134→   134→  return AVATAR_COLORS[hashString(name) % AVATAR_COLORS.length];
   135→   135→}

   137→   137→function initials(name: string | null): string {
   138→   138→  if (!name) return "?";
   139→   139→  const parts = name.trim().split(/\s+/).filter(Boolean);
   140→   140→  if (parts.length === 0) return "?";
   141→   141→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   142→   142→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   143→   143→}

  1518→  1518→function MessageList({
  1519→  1519→  messages,
  1520→  1520→  aiTyping,
  1521→  1521→}: {
  1522→  1522→  messages: Message[];
  1523→  1523→  aiTyping: boolean;
  1524→  1524→}) {
  1525→  1525→  if (messages.length === 0) {
  1526→  1526→    return (
  1527→  1527→      <div className="flex flex-col items-center justify-center h-full text-center text-muted-foreground py-16">
  1528→  1528→        <MessageSquare className="size-10 mb-3 opacity-40" />

  1533→  1533→      </div>
  1534→  1534→    );
  1535→  1535→  }
  1536→  1536→
  1537→  1537→  // Group consecutive messages by same role + < 5 min gap.
  1538→  1538→  type Group = { role: Message["role"]; items: Message[] };
  1539→  1539→  const groups: Group[] = [];
  1540→  1540→  for (const msg of messages) {
  1541→  1541→    const last = groups[groups.length - 1];
  1542→  1542→    if (
  1543→  1543→      last &&
  1544→  1544→      last.role === msg.role &&
  1545→  1545→      last.items.length > 0 &&
  1546→  1546→      new Date(msg.createdAt).getTime() -
  1547→  1547→        new Date(last.items[last.items.length - 1].createdAt).getTime() <
  1548→  1548→        5 * 60 * 1000
  1549→  1549→    ) {
  1550→  1550→      last.items.push(msg);
  1551→  1551→    } else {
  1552→  1552→      groups.push({ role: msg.role, items: [msg] });
  1553→  1553→    }
  1554→  1554→  }
  1555→  1555→
  1556→  1556→  return (
  1557→  1557→    <div className="space-y-4 max-w-2xl mx-auto">
  1558→  1558→      {groups.map((group, gi) => (
  1559→  1559→        <MessageGroup key={gi} group={group} />
  1560→  1560→      ))}
  1561→  1561→      {aiTyping && <TypingBubble />}
  1562→  1562→    </div>
  1563→  1563→  );
