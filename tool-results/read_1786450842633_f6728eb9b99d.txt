   134→   134→  try {
   135→   135→    return new Date(iso).toLocaleDateString(undefined, {
   136→   136→      year: "numeric",
   137→   137→      month: "short",
   138→   138→      day: "numeric",
   139→   139→    });
   140→   140→  } catch {
   141→   141→    return iso;
   142→   142→  }

   145→   145→function initials(name: string) {
   146→   146→  return name
   147→   147→    .split(" ")
