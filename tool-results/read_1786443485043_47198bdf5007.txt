   118→   118→function hashString(s: string): number {
   119→   119→  let h = 0;
   120→   120→  for (let i = 0; i < s.length; i++) {
   121→   121→    h = (h << 5) - h + s.charCodeAt(i);
   122→   122→    h |= 0;
   123→   123→  }
   124→   124→  return Math.abs(h);
   125→   125→}

   127→   127→function avatarColor(name: string): string {
   128→   128→  return AVATAR_COLORS[hashString(name) % AVATAR_COLORS.length];
   129→   129→}

   131→   131→function initials(name: string | null): string {
   132→   132→  if (!name) return "?";
   133→   133→  const parts = name.trim().split(/\s+/).filter(Boolean);
   134→   134→  if (parts.length === 0) return "?";
   135→   135→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   136→   136→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   137→   137→}

  1415→  1415→  // Group consecutive messages by same role + < 5 min gap.
  1416→  1416→  type Group = { role: Message["role"]; items: Message[] };
  1417→  1417→  const groups: Group[] = [];

  1423→  1423→      last.items.length > 0 &&
  1424→  1424→      new Date(msg.createdAt).getTime() -
  1425→  1425→        new Date(last.items[last.items.length - 1].createdAt).getTime() <
  1426→  1426→        5 * 60 * 1000
  1427→  1427→    ) {
