   158→function hashString(s: string): number {
   159→  let h = 0;
   160→  for (let i = 0; i < s.length; i++) {
   161→    h = (h << 5) - h + s.charCodeAt(i);
   162→    h |= 0;
   163→  }
   164→  return Math.abs(h);
   165→}
   166→
   167→function avatarColor(name: string): string {
   168→  return AVATAR_COLORS[hashString(name) % AVATAR_COLORS.length];
   169→}

   171→function initials(name: string | null): string {
   172→  if (!name) return "?";
   173→  const parts = name.trim().split(/\s+/).filter(Boolean);
   174→  if (parts.length === 0) return "?";
   175→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   176→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   177→}
