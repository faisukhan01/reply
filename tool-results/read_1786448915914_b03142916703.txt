   115→function formatDate(iso: string) {
   116→  try {
   117→    return new Date(iso).toLocaleDateString(undefined, {
   118→      year: "numeric",
   119→      month: "short",
   120→      day: "numeric",
   121→    });
   122→  } catch {
   123→    return iso;
   124→  }
   125→}

   127→function initials(name: string) {
   128→  return name
   129→    .split(" ")
   130→    .map((p) => p[0])
   131→    .filter(Boolean)
   132→    .slice(0, 2)
   133→    .join("")
   134→    .toUpperCase();
   135→}
