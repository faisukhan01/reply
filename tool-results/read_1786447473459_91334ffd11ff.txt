   110→function formatDate(iso: string) {
   111→  try {
   112→    return new Date(iso).toLocaleDateString(undefined, {
   113→      year: "numeric",
   114→      month: "short",
   115→      day: "numeric",
   116→    });
   117→  } catch {
   118→    return iso;
   119→  }
   120→}
   121→
   122→function initials(name: string) {
   123→  return name
   124→    .split(" ")
   125→    .map((p) => p[0])
   126→    .filter(Boolean)
   127→    .slice(0, 2)
   128→    .join("")
   129→    .toUpperCase();
   130→}

   903→function UsageBar({
   904→  label,
   905→  used,
   906→  limit,
   907→  display,
   908→}: {
   909→  label: string;
   910→  used: number;
   911→  limit: number;
   912→  display: string;
   913→}) {
   914→  const pct = limit > 0 ? Math.min(100, Math.round((used / limit) * 100)) : 0;
   915→  return (
   916→    <div className="space-y-1.5">
   917→      <div className="flex items-center justify-between text-sm">
   918→        <span className="text-muted-foreground">{label}</span>
   919→        <span className="font-medium">{display}</span>
   920→      </div>
   921→      <Progress value={pct} />
   922→    </div>
   923→  );
   924→}
